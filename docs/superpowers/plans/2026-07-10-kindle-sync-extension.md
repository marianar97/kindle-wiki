# Kindle Wiki Sync Chrome Extension Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Chrome extension that syncs the entire Amazon Kindle library (read.amazon.com/notebook) into `vault/raw/` in one click, replacing `sync.py`, `import_amazon.py`, and the bookmarklet.

**Architecture:** MV3 extension. Toolbar click opens a full-page sync tab. The tab fetches Amazon's own notebook endpoints (library list, then per-book annotation pages via pagination token), parses them with pure functions, renders markdown identical to `sync.py`'s format (plus highlight color, minus page numbers), and writes into the vault through the File System Access API with a persisted directory handle.

**Tech Stack:** Vanilla JS ES modules, no build step. Vitest + happy-dom for tests. `crypto.subtle` for SHA-1 block IDs.

**Spec:** `docs/superpowers/specs/2026-07-10-amazon-kindle-sync-extension-design.md`

```js
// Port of sync.py's rendering semantics. Block IDs must stay byte-compatible:
// sha1("title|type|location|text") truncated to 12 hex chars, where a missing
// location renders as "None" exactly like Python's f-string did.

export async function clipId(clip) {
  const key = `${clip.title}|${clip.type}|${clip.location ?? "None"}|${clip.text}`;
  const digest = await crypto.subtle.digest("SHA-1", new TextEncoder().encode(key));
  return [...new Uint8Array(digest)]
    .map((b) => b.toString(16).padStart(2, "0"))
    .join("")
    .slice(0, 12);
}

export function slugify(title) {
  const slug = title
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
  return slug || "untitled";
}
```

```js
export const GENERATED_NOTE =
  "<!-- GENERATED by Kindle Wiki Sync — edits are overwritten on every sync -->";

const TYPE_RANK = { highlight: 0, note: 1 };

export async function assignIds(clips) {
  return Promise.all(clips.map(async (clip) => ({ ...clip, id: await clipId(clip) })));
}

export function dedupe(clips) {
  const seen = new Set();
  const result = [];
  for (const clip of clips) {
    if (seen.has(clip.id)) continue;
    seen.add(clip.id);
    result.push(clip);
  }
  return result;
}

function locationStart(clip) {
  if (!clip.location) return 0;
  return parseInt(clip.location.split("-")[0], 10) || 0;
}

export function sortClips(clips) {
  return [...clips].sort(
    (a, b) => locationStart(a) - locationStart(b) || TYPE_RANK[a.type] - TYPE_RANK[b.type]
  );
}

function metaLine(clip) {
  const parts = [clip.type];
  if (clip.type === "highlight" && clip.color) parts.push(clip.color);
  if (clip.location) parts.push(`loc. ${clip.location}`);
  return parts.join(" · ");
}

function renderClip(clip) {
  if (clip.type === "highlight") {
    const quoted = clip.text.split("\n").map((line) => `> ${line}`).join("\n");
    return `${quoted}\n>\n> — ${metaLine(clip)} ^${clip.id}`;
  }
  return `Note: ${clip.text}\n— ${metaLine(clip)} ^${clip.id}`;
}

export function renderBook(title, author, clips, syncedOn) {
  const lines = [
    "---",
    `title: ${JSON.stringify(title)}`,
    `author: ${author ? JSON.stringify(author) : "null"}`,
    `last_synced: ${syncedOn}`,
    "---",
    GENERATED_NOTE,
    "",
    `# ${title}`,
  ];
  if (author) lines.push(`*${author}*`);
  lines.push("", "## Highlights", "");
  const body = clips.map(renderClip).join("\n\n");
  return `${lines.join("\n")}\n${body}\n`;
}
```

Note: `JSON.stringify` does not escape non-ASCII characters, matching Python's `json.dumps(..., ensure_ascii=False)`.

```js
// Parsers for read.amazon.com/notebook markup. Pure functions: HTML in,
// data out. Selectors verified against the live DOM on 2026-07-10.

function clean(value) {
  return (value || "").replace(/\s+/g, " ").trim();
}

function parseDoc(html) {
  return new DOMParser().parseFromString(html, "text/html");
}

export function parseLibrary(html) {
  const doc = parseDoc(html);
  const books = [];
  for (const el of doc.querySelectorAll("#kp-notebook-library .kp-notebook-library-each-book")) {
    const title = clean(el.querySelector("h2")?.textContent);
    if (!el.id || !title) continue;
    const author = clean(el.querySelector("p")?.textContent).replace(/^by:?\s*/i, "");
    books.push({ asin: el.id, title, author: author || null });
  }
  return books;
}

function parseHeader(row) {
  const header = clean(row.querySelector("#annotationHighlightHeader")?.textContent);
  const color = header.match(/^([a-z]+) highlight/i)?.[1]?.toLowerCase() ?? null;
  const headerLocation = header.match(/location:\s*([\d,–-]+)/i)?.[1]?.replace(/,/g, "") ?? null;
  return { color, headerLocation };
}

export function parseAnnotations(html) {
  const doc = parseDoc(html);
  const clips = [];
  for (const row of doc.querySelectorAll("#kp-notebook-annotations .a-row.a-spacing-base")) {
    const text = clean(row.querySelector("#highlight")?.textContent);
    const note = clean(row.querySelector("#note")?.textContent);
    if (!text && !note) continue;
    const { color, headerLocation } = parseHeader(row);
    const location = clean(row.querySelector("#kp-annotation-location")?.value) || headerLocation;
    if (text) clips.push({ type: "highlight", text, location, color });
    if (note) clips.push({ type: "note", text: note, location, color: null });
  }
  return {
    clips,
    nextToken: doc.querySelector(".kp-notebook-annotations-next-page-start")?.value || null,
    limitState: doc.querySelector(".kp-notebook-content-limit-state")?.value ?? "",
  };
}

export function isLoginHtml(html) {
  return /action="[^"]*\/ap\/signin|name="signIn"/i.test(html);
}
```

```js
// Vault access: File System Access API writes plus the pure decision logic
// around them. The directory handle for vault/ persists in IndexedDB.

const DB_NAME = "kindle-wiki-sync";
const STORE = "handles";
const VAULT_KEY = "vault";
const BLOCK_ID_RE = /\^([0-9a-f]{12})/g;

// --- pure logic -------------------------------------------------------------

export function stripVolatile(text) {
  return text.replace(/^last_synced: .*$/m, "last_synced:");
}

export function shouldWrite(oldContent, newContent) {
  if (oldContent == null) return true;
  return stripVolatile(oldContent) !== stripVolatile(newContent);
}

export function countNewClips(oldContent, clips) {
  const oldIds = new Set((oldContent ?? "").match(BLOCK_ID_RE)?.map((m) => m.slice(1)) ?? []);
  return clips.filter((c) => !oldIds.has(c.id)).length;
}

export function snapshotName(date) {
  const pad = (n) => String(n).padStart(2, "0");
  const stamp =
    `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}` +
    `-${pad(date.getHours())}${pad(date.getMinutes())}${pad(date.getSeconds())}`;
  return `${stamp}-amazon-notebook.json`;
}

// --- IndexedDB handle persistence -------------------------------------------

function openDb() {
  return new Promise((resolve, reject) => {
    const req = indexedDB.open(DB_NAME, 1);
    req.onupgradeneeded = () => req.result.createObjectStore(STORE);
    req.onsuccess = () => resolve(req.result);
    req.onerror = () => reject(req.error);
  });
}

async function dbOp(mode, fn) {
  const db = await openDb();
  try {
    return await new Promise((resolve, reject) => {
      const tx = db.transaction(STORE, mode);
      const req = fn(tx.objectStore(STORE));
      req.onsuccess = () => resolve(req.result);
      req.onerror = () => reject(req.error);
    });
  } finally {
    db.close();
  }
}

export async function saveVaultHandle(handle) {
  await dbOp("readwrite", (store) => store.put(handle, VAULT_KEY));
}

export async function loadVaultHandle() {
  return (await dbOp("readonly", (store) => store.get(VAULT_KEY))) ?? null;
}

export async function clearVaultHandle() {
  await dbOp("readwrite", (store) => store.delete(VAULT_KEY));
}

// --- File System Access ------------------------------------------------------

export async function pickVault() {
  const handle = await window.showDirectoryPicker({ id: "kindle-wiki-vault", mode: "readwrite" });
  await saveVaultHandle(handle);
  return handle;
}

export async function ensurePermission(handle) {
  const opts = { mode: "readwrite" };
  if ((await handle.queryPermission(opts)) === "granted") return true;
  return (await handle.requestPermission(opts)) === "granted";
}

async function getDir(vaultHandle, names) {
  let dir = vaultHandle;
  for (const name of names) dir = await dir.getDirectoryHandle(name, { create: true });
  return dir;
}

async function readFileIfExists(dir, name) {
  try {
    const file = await (await dir.getFileHandle(name)).getFile();
    return await file.text();
  } catch (err) {
    if (err.name === "NotFoundError") return null;
    throw err;
  }
}

async function writeFile(dir, name, content) {
  const handle = await dir.getFileHandle(name, { create: true });
  const writable = await handle.createWritable();
  await writable.write(content);
  await writable.close();
}

// Writes raw/books/<slug>.md unless unchanged. Returns clip/new-clip counts.
export async function writeBook(vaultHandle, slug, content, clips) {
  const dir = await getDir(vaultHandle, ["raw", "books"]);
  const name = `${slug}.md`;
  const old = await readFileIfExists(dir, name);
  if (!shouldWrite(old, content)) return { written: false, newClips: 0 };
  await writeFile(dir, name, content);
  return { written: true, newClips: countNewClips(old, clips) };
}

// Archives the full-library JSON snapshot to raw/clippings/.
export async function archiveSnapshot(vaultHandle, payload, date = new Date()) {
  const dir = await getDir(vaultHandle, ["raw", "clippings"]);
  await writeFile(dir, snapshotName(date), JSON.stringify(payload, null, 2));
}
```

```js
// Orchestrates a full sync: library -> per-book clips -> render -> write.
// Dependencies are injected so tests can fake the network and filesystem.

import { assignIds, dedupe, renderBook, slugify, sortClips } from "./renderer.js";

function today() {
  const d = new Date();
  const pad = (n) => String(n).padStart(2, "0");
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;
}

function uniqueSlug(title, author, used) {
  let slug = slugify(title);
  if (used.has(slug)) slug = `${slug}-${slugify(author || "unknown")}`;
  used.add(slug);
  return slug;
}

export async function runSync(deps, vaultHandle, onProgress) {
  const { fetchLibrary, fetchBookClips, writeBook, archiveSnapshot } = deps;
  const syncedOn = today();
  const books = await fetchLibrary();
  onProgress({ kind: "library", count: books.length });

  const summary = { books: books.length, booksWritten: 0, newClips: 0, failures: [] };
  const snapshot = { exported: new Date().toISOString(), books: [] };
  const usedSlugs = new Set();

  for (const book of books) {
    onProgress({ kind: "book-start", title: book.title });
    try {
      const rawClips = await fetchBookClips(book.asin);
      snapshot.books.push({ ...book, clips: rawClips });
      if (rawClips.length === 0) {
        onProgress({ kind: "book-done", title: book.title, clips: 0, written: false });
        continue;
      }
      const clips = sortClips(
        dedupe(
          await assignIds(
            rawClips.map((c) => ({ ...c, title: book.title, author: book.author }))
          )
        )
      );
      const slug = uniqueSlug(book.title, book.author, usedSlugs);
      const content = renderBook(book.title, book.author, clips, syncedOn);
      const result = await writeBook(vaultHandle, slug, content, clips);
      if (result.written) summary.booksWritten += 1;
      summary.newClips += result.newClips;
      onProgress({ kind: "book-done", title: book.title, clips: clips.length, written: result.written });
    } catch (err) {
      summary.failures.push({ title: book.title, error: err.message });
      onProgress({ kind: "book-failed", title: book.title, error: err.message });
    }
  }

  await archiveSnapshot(vaultHandle, snapshot);
  onProgress({ kind: "done", summary });
  return summary;
}
```

Note the slug-collision difference from `sync.py`: `sync.py` suffixed with the author slug *or the first clip's id*; here books always have a title and we fall back to `-unknown` when the author is missing. Same observable behavior for real libraries.
